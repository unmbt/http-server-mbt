cmake_minimum_required(VERSION 3.15)
project(hs_libevent_example C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(EVENT__DISABLE_OPENSSL ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_REGRESS ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_SAMPLES ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_BENCHMARK ON CACHE BOOL "" FORCE)
set(EVENT__LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory("${LIBEVENT_SOURCE}" libevent)
foreach(mode static dynamic)
  add_executable(libevent_${mode} libevent_server.c)
  target_include_directories(libevent_${mode} PRIVATE ../../c_abi/include)
  target_link_libraries(libevent_${mode} PRIVATE event_extra_static event_core_static)
  if(mode STREQUAL static)
    target_compile_definitions(libevent_${mode} PRIVATE HS_STATIC)
    target_link_libraries(libevent_${mode} PRIVATE "${HS_ARCHIVE}")
  else()
    target_compile_definitions(libevent_${mode} PRIVATE HS_EXAMPLE_DYNAMIC)
  endif()
  if(WIN32)
    target_link_libraries(libevent_${mode} PRIVATE ws2_32 iphlpapi bcrypt crypt32 secur32 mswsock userenv advapi32 synchronization dbghelp shell32)
    target_link_options(libevent_${mode} PRIVATE /NODEFAULTLIB:msvcrt.lib)
  else()
    target_link_libraries(libevent_${mode} PRIVATE pthread m ${CMAKE_DL_LIBS})
  endif()
endforeach()
